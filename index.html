function valueAtPixelNearestValid(latlng, r=2){ const R = currentData ? currentData._render : null; if (!R || !map) return null; // Bounds del overlay (lat/lon) pero el cálculo de (x,y) lo hacemos en PROYECCIÓN const b = R.bounds; const nw = b.getNorthWest(); const se = b.getSouthEast(); // Coordenadas en pixeles de capa (WebMercator) -> coincide con lo que se ve const pNW = map.latLngToLayerPoint(nw); const pSE = map.latLngToLayerPoint(se); const p = map.latLngToLayerPoint(latlng); const dx = (pSE.x - pNW.x); const dy = (pSE.y - pNW.y); if (dx === 0 || dy === 0) return null; const x = (p.x - pNW.x) / dx; // 0..1 izquierda->derecha const y = (p.y - pNW.y) / dy; // 0..1 arriba->abajo if (x < 0 || x > 1 || y < 0 || y > 1) return null; // Pixel en la imagen (top-left) const xPix = x * (R.nx - 1); const yPix = y * (R.ny - 1); // Convertir a índice del grid original (i,j) const iFloat = R.latIncreasing ? ((R.ny - 1) - yPix) : yPix; const jFloat = xPix; const i0 = Math.min(R.ny - 1, Math.max(0, Math.round(iFloat))); const j0 = Math.min(R.nx - 1, Math.max(0, Math.round(jFloat))); // primero intenta exacto let v = R.z[i0][j0]; if (v != null && isFinite(v)) return v; // busca vecino válido cercano (quita “huecos” por máscara) for (let di = -r; di <= r; di++){ for (let dj = -r; dj <= r; dj++){ const i = i0 + di, j = j0 + dj; if (i<0 || j<0 || i>=R.ny || j>=R.nx) continue; v = R.z[i][j]; if (v != null && isFinite(v)) return v; } } return null; }
